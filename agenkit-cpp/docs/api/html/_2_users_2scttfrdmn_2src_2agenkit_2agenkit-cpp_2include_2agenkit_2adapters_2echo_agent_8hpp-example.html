Simple agent that echoes back the input messageThis is a reference implementation of the Agent interface, useful for testing and examples.

EchoAgent agent;
auto msg = Message::with_text("user", "Hello!");
auto future = agent.process(std::move(msg));
auto result = future.get();
if (result.is_ok()) {
std::cout << "Response: " << result.unwrap().content_as_str() << std::endl;
}
