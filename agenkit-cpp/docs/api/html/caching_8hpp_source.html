137 struct Snapshot {
138 uint64_t total_requests;
139 uint64_t cache_hits;
140 uint64_t cache_misses;
141 uint64_t cache_evictions;
142 uint64_t cache_invalidations;
143 uint64_t expired_entries;
144 size_t current_cache_size;
145 double hit_rate;
146 double miss_rate;
147 };
148
149 Snapshot snapshot(size_t current_cache_size) const {
150 auto total = total_requests.load();
151 auto hits = cache_hits.load();
152 auto misses = cache_misses.load();
153
154 return Snapshot{
155 total,
156 hits,
157 misses,
158 cache_evictions.load(),
159 cache_invalidations.load(),
160 expired_entries.load(),
161 current_cache_size,
162 total > 0 ? static_cast<double>(hits) / total : 0.0,
163 total > 0 ? static_cast<double>(misses) / total : 0.0
164 };
165 }
166};

169class CachingMiddleware : public core::Agent {
170public:
175 CachingMiddleware(
176 std::shared_ptr<core::Agent> agent,
177 CachingConfig config = CachingConfig()
178 ) : agent_(std::move(agent)), config_(std::move(config)) {
179 config_.validate();
180
181 // Use default key generator if none provided
182 if (!config_.key_generator) {
183 config_.key_generator = [](const core::Message& msg) {
184 return std::to_string(std::hash<std::string>{}(msg.content_as_str()));
185 };
186 }
187 }
188
189 std::string name() const override {
190 return "caching(" + agent_->name() + ")";
191 }
192
193 std::future<core::Result<core::Message, core::AgentError>>
194 process(core::Message message) override;
195
197 const CachingMetrics& metrics() const {
198 return metrics_;
199 }
200
202 const CachingConfig& config() const {
203 return config_;
204 }
205
207 size_t cache_size() const {
208 std::shared_lock<std::shared_mutex> lock(mutex_);
209 return cache_.size();
210 }
211
213 void clear_cache() {
214 std::unique_lock<std::shared_mutex> lock(mutex_);
215 cache_.clear();
216 lru_list_.clear();
217 }
218
223 bool invalidate(const core::Message& message) {
224 std::unique_lock<std::shared_mutex> lock(mutex_);
225 auto key = config_.key_generator(message);
226 auto it = cache_.find(key);
227 if (it != cache_.end()) {
228 lru_list_.erase(it->second.second);
229 cache_.erase(it);
230 metrics_.cache_invalidations++;
231 return true;
232 }
233 return false;
234 }
235
236private:
237 std::shared_ptr<core::Agent> agent_;
238 CachingConfig config_;
239 mutable CachingMetrics metrics_;
240
241 // Cache storage (protected by shared_mutex)
242 mutable std::shared_mutex mutex_;
243 std::unordered_map<
244 std::string,
245 std::pair<CacheEntry, std::list<std::string>::iterator>
246 > cache_;
247 std::list<std::string> lru_list_; // Most recently used at front
248
250 std::optional<core::Message> get_cached(const std::string& key);
251
