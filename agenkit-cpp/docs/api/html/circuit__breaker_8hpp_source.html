135struct CircuitBreakerMetrics {
136 std::atomic<uint64_t> total_requests{0};

138 std::atomic<uint64_t> failed_requests{0};

144 struct Snapshot {
145 uint64_t total_requests;
146 uint64_t successful_requests;
147 uint64_t failed_requests;
148 uint64_t rejected_requests;
149 uint64_t state_transitions;
150 uint64_t last_state_change_ms;
151 CircuitState current_state;
152 double success_rate;
153 double rejection_rate;
154 };
155
156 Snapshot snapshot(CircuitState current_state) const {
157 auto total = total_requests.load();
158 auto success = successful_requests.load();
159 auto rejected = rejected_requests.load();
160
161 return Snapshot{
162 total,
163 success,
164 failed_requests.load(),
165 rejected,
166 state_transitions.load(),
167 last_state_change_ms.load(),
168 current_state,
169 total > 0 ? static_cast<double>(success) / total : 0.0,
170 total > 0 ? static_cast<double>(rejected) / total : 0.0
171 };
172 }
173};
