43class ObjectPool {
44public:

53 ~ObjectPool() {

55
56 // All objects should be released before pool destruction
57 // Free list contains pointers, not ownership
58 free_list_.clear();
59
60 // Blocks own the memory, will be freed automatically

78 template<typename... Args>
79 T* acquire(Args&&... args) {
80 std::lock_guard<std::mutex> lock(mutex_);
81
82 // If free list is empty, allocate a new block
83 if (free_list_.empty()) {
84 allocate_block();
85 }
86
87 // Get object from free list
88 T* obj = free_list_.back();
89 free_list_.pop_back();
90
91 // Construct object in-place using placement new
92 new (obj) T(std::forward<Args>(args)...);
93
94 return obj;
95 }
96
105 void release(T* obj) {
106 if (!obj) {
107 return;
108 }
109
110 std::lock_guard<std::mutex> lock(mutex_);
111
112 // Call destructor
113 obj->~T();
114
115 // Return to free list
116 free_list_.push_back(obj);
117 }
118
123 size_t available() const {
124 std::lock_guard<std::mutex> lock(mutex_);
125 return free_list_.size();
126 }
127
132 size_t capacity() const {
133 std::lock_guard<std::mutex> lock(mutex_);
134 return blocks_.size() * BlockSize;
135 }
136
137private:
144 void allocate_block() {
145 // Allocate raw memory for BlockSize objects
146 auto block = std::make_unique<uint8_t[]>(sizeof(T) * BlockSize);
147
148 // Add pointers to all objects in this block to free list
149 uint8_t* ptr = block.get();
150 for (size_t i = 0; i < BlockSize; i++) {
151 free_list_.push_back(reinterpret_cast<T*>(ptr + i * sizeof(T)));
152 }
153
154 // Store block to keep memory alive
155 blocks_.push_back(std::move(block));
156 }
157
158 std::vector<std::unique_ptr<uint8_t[]>> blocks_;
159 std::vector<T*> free_list_;
160 mutable std::mutex mutex_;
161};
