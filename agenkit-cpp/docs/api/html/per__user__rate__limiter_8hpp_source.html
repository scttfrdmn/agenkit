107 void validate() const {
108 if (rate_per_second <= 0.0) {
109 throw std::invalid_argument("rate_per_second must be positive");
110 }
111 if (capacity < 1) {
112 throw std::invalid_argument("capacity must be >= 1");
113 }
114 if (tokens_per_request < 1) {
115 throw std::invalid_argument("tokens_per_request must be >= 1");
116 }
117 if (tokens_per_request > capacity) {
118 throw std::invalid_argument("tokens_per_request must be <= capacity");
119 }
120 if (max_wait_timeout.has_value() && max_wait_timeout->count() <= 0) {
121 throw std::invalid_argument("max_wait_timeout must be positive if set");
122 }
123 if (!user_id_extractor) {
124 throw std::invalid_argument("user_id_extractor must be set");
125 }
126 }

178struct PerUserRateLimiterMetrics {
179 std::atomic<uint64_t> total_requests{0};
180 std::atomic<uint64_t> allowed_requests{0};
181 std::atomic<uint64_t> rejected_requests{0};
182 std::atomic<uint64_t> total_wait_time_ms{0};
183 std::atomic<size_t> active_users{0};
184
186 struct Snapshot {
187 uint64_t total_requests;
188 uint64_t allowed_requests;
189 uint64_t rejected_requests;
190 uint64_t total_wait_time_ms;
191 size_t active_users;
192 double rejection_rate;
193 double avg_wait_time_ms;
194 };
195
196 Snapshot snapshot() const {
197 auto total = total_requests.load();
198 auto rejected = rejected_requests.load();
199 auto wait_time = total_wait_time_ms.load();
200
201 return Snapshot{
202 total,
203 allowed_requests.load(),
204 rejected,
205 wait_time,
206 active_users.load(),
207 total > 0 ? static_cast<double>(rejected) / total : 0.0,
208 total > 0 ? static_cast<double>(wait_time) / total : 0.0
209 };
210 }
211};
