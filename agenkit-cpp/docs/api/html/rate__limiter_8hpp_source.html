145struct RateLimiterMetrics {
146 std::atomic<uint64_t> total_requests{0};
147 std::atomic<uint64_t> allowed_requests{0};
148 std::atomic<uint64_t> rejected_requests{0};
149 std::atomic<uint64_t> waited_requests{0};
150 std::atomic<uint64_t> total_wait_time_ms{0};
151
153 struct Snapshot {
154 uint64_t total_requests;
155 uint64_t allowed_requests;
156 uint64_t rejected_requests;
157 uint64_t waited_requests;
158 uint64_t total_wait_time_ms;
159 double current_tokens;
160 double rejection_rate;
161 double avg_wait_time_ms;
162 };
163
164 Snapshot snapshot(double current_tokens) const {
165 auto total = total_requests.load();
166 auto rejected = rejected_requests.load();
167 auto waited = waited_requests.load();
168 auto wait_time = total_wait_time_ms.load();
169
170 return Snapshot{
171 total,
172 allowed_requests.load(),
173 rejected,
174 waited,
175 wait_time,
176 current_tokens,
177 total > 0 ? static_cast<double>(rejected) / total : 0.0,
178 waited > 0 ? static_cast<double>(wait_time) / waited : 0.0
179 };
180 }
181};
