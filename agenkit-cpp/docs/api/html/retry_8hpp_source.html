133struct RetryMetrics {
134 std::atomic<uint64_t> total_attempts{0};
135 std::atomic<uint64_t> total_retries{0};
136 std::atomic<uint64_t> successful_on_retry{0};
137 std::atomic<uint64_t> failed_after_retries{0};
138 std::atomic<uint64_t> total_backoff_ms{0};
139
141 struct Snapshot {
142 uint64_t total_attempts;
143 uint64_t total_retries;
144 uint64_t successful_on_retry;
145 uint64_t failed_after_retries;
146 uint64_t total_backoff_ms;
147 double avg_retries_per_request;
148 double success_rate_after_retry;
149 };
150
151 Snapshot snapshot() const {
152 auto attempts = total_attempts.load();
153 auto retries = total_retries.load();
154 auto success = successful_on_retry.load();
155 auto failed = failed_after_retries.load();
156
157 return Snapshot{
158 attempts,
159 retries,
160 success,
161 failed,
162 total_backoff_ms.load(),
163 attempts > 0 ? static_cast<double>(retries) / attempts : 0.0,
164 retries > 0 ? static_cast<double>(success) / retries : 0.0
165 };
166 }
167};
