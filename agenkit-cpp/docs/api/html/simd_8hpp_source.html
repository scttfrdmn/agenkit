46inline double horizontal_sum_pd(__m256d vec) {
47 // Horizontal add: [a,b,c,d] -> [a+b, c+d, a+b, c+d]
48 __m256d hadd = _mm256_hadd_pd(vec, vec);
49
50 // Extract high 128 bits
51 __m128d high = _mm256_extractf128_pd(hadd, 1);
52
53 // Extract low 128 bits
54 __m128d low = _mm256_castpd256_pd128(hadd);
55
56 // Add high and low
57 __m128d sum128 = _mm_add_pd(high, low);
58
59 // Extract result
60 return _mm_cvtsd_f64(sum128);
61}

116inline double variance_avx2(const double* values, size_t size, double mean) {
117 __m256d mean_vec = _mm256_set1_pd(mean);
118 __m256d variance_vec = _mm256_setzero_pd();
119
120 size_t i = 0;
121
122 // Process 4 elements at a time
123 for (; i + 4 <= size; i += 4) {
124 __m256d vals = _mm256_loadu_pd(&values[i]);
125 __m256d diff = _mm256_sub_pd(vals, mean_vec);
126 __m256d sq = _mm256_mul_pd(diff, diff);
127 variance_vec = _mm256_add_pd(variance_vec, sq);
128 }
129
130 // Horizontal sum of variance_vec
131 double variance = horizontal_sum_pd(variance_vec);
132
133 // Handle remaining elements (scalar)
134 for (; i < size; i++) {
135 double diff = values[i] - mean;
136 variance += diff * diff;
137 }
138
139 return variance / static_cast<double>(size);
140}

151inline double mean_avx2(const double* values, size_t size) {
152 if (size == 0) return 0.0;
153
154 __m256d sum_vec = _mm256_setzero_pd();
155
156 size_t i = 0;
157
158 // Process 4 elements at a time
159 for (; i + 4 <= size; i += 4) {
160 __m256d vals = _mm256_loadu_pd(&values[i]);
161 sum_vec = _mm256_add_pd(sum_vec, vals);
162 }
163
164 // Horizontal sum
165 double sum = horizontal_sum_pd(sum_vec);
166
167 // Handle remaining elements (scalar)
168 for (; i < size; i++) {
169 sum += values[i];
170 }
171
172 return sum / static_cast<double>(size);
173}

187inline double calculate_variance(const double* values, size_t size, double mean) {
188#if AGENKIT_HAS_AVX2
