144template<typename F, typename... Args>
145auto ThreadPool::enqueue(F&& f, Args&&... args)
146 -> std::future<typename std::invoke_result<F, Args...>::type>
147{
148 using return_type = typename std::invoke_result<F, Args...>::type;
149
150 // Create packaged task (wraps function and provides future)
151 auto task = std::make_shared<std::packaged_task<return_type()>>(
152 std::bind(std::forward<F>(f), std::forward<Args>(args)...)
153 );
154
155 std::future<return_type> result = task->get_future();
156
157 {
158 std::unique_lock<std::mutex> lock(queue_mutex_);
159
160 // Don't allow enqueueing after stopping the pool
161 if (stop_) {
162 throw std::runtime_error("Cannot enqueue task on stopped ThreadPool");
163 }
164
165 // Add task to queue
166 tasks_.emplace([task]() { (*task)(); });
167 }
168
169 // Notify one waiting thread
170 condition_.notify_one();
171
172 return result;
173}
