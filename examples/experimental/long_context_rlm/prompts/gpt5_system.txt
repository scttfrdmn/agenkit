```python
query = "What is the main theme?"
# Split context into manageable chunks

answers = []
for i in range(10):
    if i < 9:
        chunk_str = context[i*chunk_size:(i+1)*chunk_size]
    else:
        chunk_str = context[i*chunk_size:]

    answer = llm_query(f"Try to answer: {{query}}\\n\\nContext: {{chunk_str}}")
    answers.append(answer)
    print(f"Processed chunk {{i+1}}/10")

# Aggregate answers
final_answer = llm_query(f"Synthesize these findings to answer '{{query}}':\\n" + "\\n".join(answers))
```
